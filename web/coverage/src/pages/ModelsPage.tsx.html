  const [draft, setDraft] = useState<ModelSource[]>(sources);
  const [saving, setSaving] = useState(false);
 
  useEffect(() => { setDraft(sources); }, [sources]);
 
  const update = (i: number, patch: Partial<ModelSource>) =>
